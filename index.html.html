    async function run() {
      // Start background music after boot sound finishes
      setTimeout(() => {
        const bgm = document.getElementById('bg-music');

      }, 3500);


      const urls = [
        "https://cdn.jsdelivr.net/gh/boibn/120-1@main/EaglercraftX_1.20.6_WASM-GC_Offline_Download.html.part1",
        "https://cdn.jsdelivr.net/gh/boibn/120-2@main/EaglercraftX_1.20.6_WASM-GC_Offline_Download.html.part2",


"https://cdn.jsdelivr.net/gh/boibn/120-3@main/EaglercraftX_1.20.6_WASM-GC_Offline_Download.html.part3"
      ];
     
      let fullHtml = "";
      try {
        for (const u of urls) {
          const res = await fetch(u);
          fullHtml += await res.text();
        }
      } catch (e) {
        console.error("Fetch error:", e);
      }


      // Fixed loading screen time of 3.5 seconds
      setTimeout(() => {
        const ls = document.getElementById('loading-screen');
        if (ls) ls.style.opacity = '0';
       
        setTimeout(() => {
          // Attempt to open the game in a fresh, final window
          const gameWin = window.open('', '_blank', 'width=' + screen.availWidth + ',height=' + screen.availHeight + ',menubar=no,status=no,toolbar=no');
         
          if (gameWin) {
            gameWin.document.open();
            gameWin.document.write(fullHtml);
            gameWin.document.close();
            // Close this intermediate loading window
            window.close();
          } else {
            // Safety Fallback: Load in THIS window if the pop-up is blocked
            document.open();
            document.write(fullHtml);
            document.close();
          }
        }, 800);
      }, 3500);
    }


    run();


    // Sound trigger for modern browsers
    (function() {
        const boot = document.getElementById('boot-audio');
        const trigger = () => {
            if (boot) boot.play().catch(()=>{});
            ['mousemove', 'mousedown', 'keydown', 'touchstart'].forEach(e =>
                window.removeEventListener(e, trigger)
            );
        };
        ['mousemove', 'mousedown', 'keydown', 'touchstart'].forEach(e =>
            window.addEventListener(e, trigger)
        );
        if (boot) boot.play().catch(()=>{});
    })();
